function checkPass(){
if(pass.value==="14.25"){
login.classList.add("hidden");
main.classList.remove("hidden");
}else if(pass.value==="2016"){
login.classList.add("hidden");
owner.classList.remove("hidden");
loadMsgs();
}else{
error.innerText="Galat hai… phir socho 😉";
}}

function openCard(id){
main.classList.add("hidden");
document.getElementById(id).classList.remove("hidden");
}
function goBack(){
document.querySelectorAll(".card").forEach(c=>c.classList.add("hidden"));
main.classList.remove("hidden");
}
function saveMsg(){
let arr=JSON.parse(localStorage.getItem("msgs")||"[]");
arr.push(msg.value);
localStorage.setItem("msgs",JSON.stringify(arr));
msg.value="";
sent.innerText="Tumhari baat safely pahunch gayi 💙";
}
function loadMsgs(){
let arr=JSON.parse(localStorage.getItem("msgs")||"[]");
allMsgs.innerHTML=arr.map(m=>"<p>• "+m+"</p>").join("");
}
function toggleNight(){
document.body.classList.toggle("night");
}
